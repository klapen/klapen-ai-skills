A cache-through layer is a piece of glue that sits between a service and its slow backing store: on a read, it checks the cache first, and only falls through to the database on a miss. Before this fictional PR: the endpoint read straight from the DB every time, with no cache at all. Every request paid a full round-trip. That's fine at 10 QPS. It's not fine at 10K QPS. Ut enim ad minim veniam.

Three concerns are wired together: a Redis client for the primary cache, an in-memory LRU for the fallback path when Redis is down, and a small resolver that keeps them consistent. The sequence above shows the read path; write paths (not diagrammed) always populate Redis first, then LRU.

class CacheThroughMiddleware:
    def __init__(self, redis: Redis, lru: LRUCache, schema_version: str) -> None:
        self.redis = redis
        self.lru = lru
        self.schema_version = schema_version

    async def get(self, widget_id: str) -> Widget | None:
        key = f"widget:{widget_id}:{self.schema_version}"
        try:
            raw = await self.redis.get(key)
        except RedisConnectionError:
            return self.lru.get(key)
        if raw is None:
            return None
        try:
            return Widget.model_validate_json(raw)
        except ValidationError:
            # Corrupt payload: treat as miss and evict aggressively.
            await self.redis.delete(key)
            return None

@router.get("/widgets/{widget_id}")
async def get_widget(
    widget_id: str,
    cache: CacheThroughMiddleware = Depends(get_cache),
) -> Widget:
    cached = await cache.get(widget_id)
    if cached is not None:
        return cached
    widget = await repository.get(widget_id)
    if widget is None:
        raise HTTPException(404)
    await cache.set(widget_id, widget, ttl=300)
    return widget
